81 template <typename F>
82 typename std::enable_if<std::is_void<typename std::invoke_result<F>::type>::value, bool>::type invoke(F function, bool sync = false) const {
83 if (!sync) {
84 AbstractTask *_t = new Invocable<void()>::Function(std::forward<F>(function));
85 if (!invokeTask(_t)) {
86 delete _t;
87 return false;
88 }
89 return true;
90 }
91 if (std::this_thread::get_id() == id()) {
92 processTasks();
93 function();
94 return true;
95 }
96 std::atomic_flag finished;
97 AbstractTask *_t = new Invocable<void()>::Function([&function, &finished]() {
98 function();
99 finished.test_and_set();
100 finished.notify_one();
101 });
102 if (!invokeTask(_t)) {
103 delete _t;
104 return false;
105 }
106 for (;;) {
107 finished.wait(false);
108 if (finished.test()) break;
109 }
110 return true;
111 }
