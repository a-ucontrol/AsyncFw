45 SharedLockData<T &, M> acquire() {
46 int64_t expected = expire_.load(std::memory_order_relaxed);
47 if (expected != 0 && expected <= MILLISECONDS_SINCE_EPOCH) {
48 if (expire_.compare_exchange_strong(expected, 0, std::memory_order_relaxed)) {
